NEW CODE (Multi-region support):
--------------------------------------------------
class TextToSpeechEngine:
    def __init__(self, voice="en-US-AriaNeural", preferred_engine="auto"):
        self.voice = voice
        self.preferred_engine = preferred_engine
        self.active_engine = None
        
        if preferred_engine == "auto":
            self._detect_working_engine()  # ← NEW
    
    def text_to_audio_file(self, text, output_path):
        # Try Edge TTS first
        if self.active_engine == "edge":
            try:
                self._text_to_audio_edge(text, output_path)
                return  # ✓ Success
            except:
                self.active_engine = "gtts"  # ← Auto-switch
        
        # Try Google TTS
        if self.active_engine == "gtts":
            try:
                self._text_to_audio_gtts(text, output_path)
                return  # ✓ Success
            except:
                self.active_engine = "pyttsx3"  # ← Auto-switch
        
        # Try offline TTS
        if self.active_engine == "pyttsx3":
            self._text_to_audio_pyttsx3(text, output_path)
            return  # ✓ Success

KEY DIFFERENCES:
  ✅ Automatic fallback (no manual intervention)
  ✅ Tries multiple engines
  ✅ Remembers which engine works
  ✅ Detailed error messages
  ✅ Backward compatible (old code still works)
